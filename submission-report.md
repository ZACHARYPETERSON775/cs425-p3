## Design

This project was designed and implimented using AI.

CIDR addresses are parsed in `cidr_parse()` which results in postfixs rangin from 0 to 32. The postfixes `/0`, `/31`, and `/32` are special cases. These cases affect `cidr_mask()`, `cidr_host_count()`, `cidr_first_host()`, and `cidr_last_host()`. `cidr_mask()` alwasy return an empty mask for `/0` and a full mask for `/32` or greater. `cidr_host_count()` returns 64-bit values to accompidate `/0` while detecting `/31` and `/32` and return their known host counts. `cidr_first_host()` and `cidr_last_host()` detect `/31` and `/32` and treat them as range exeptions.

`table_lookup()` scans every route in the table and checks whether is contains the desitination address using `cidr_contains()`. If a route matches, it keep if its prefix is the longest encounted. Previous matches with sorter prefixs are discarded so that results do not depend on insertion order.

#define _POSIX_C_SOURCE 200809L
#include "lab.h"
#include <ctype.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct fwd_table {
    route_t *routes;
    size_t count;
    size_t capacity;
};

static void skip_whitespace(const char **text)
{
    while (isspace((unsigned char)**text)) {
        (*text)++;
    }
}

static bool read_token(const char **text, char *token, size_t token_size)
{
    const char *cursor = *text;
    size_t length = 0;

    skip_whitespace(&cursor);
    if (*cursor == '\0') {
        return false;
    }

    while (*cursor != '\0' && !isspace((unsigned char)*cursor)) {
        if (length + 1 >= token_size) {
            return false;
        }
        token[length++] = *cursor++;
    }
    token[length] = '\0';
    *text = cursor;
    return true;
}

bool route_parse(const char *line, route_t *route)
{
    if (line == NULL || route == NULL) {
        return false;
    }

    char destination[CIDR_STR_LEN];
    char gateway[IP_STR_LEN];
    route_t parsed = { 0 };
    const char *cursor = line;

    // Read exactly three fields; comments are stripped by table_load().
    if (!read_token(&cursor, destination, sizeof(destination)) ||
        !read_token(&cursor, gateway, sizeof(gateway)) ||
        !read_token(&cursor, parsed.dev, sizeof(parsed.dev))) {
        return false;
    }
    skip_whitespace(&cursor);
    if (*cursor != '\0') {
        return false;
    }

    if (strcmp(destination, "default") == 0) {
        parsed.dest = (cidr_t){ 0, 0 };
    } else if (!cidr_parse(destination, &parsed.dest) ||
               !cidr_is_network(parsed.dest)) {
        return false;
    }

    if (strcmp(gateway, "direct") != 0 &&
        !ip_parse(gateway, &parsed.gateway)) {
        return false;
    }

    *route = parsed;
    return true;
}

fwd_table_t *table_create(void)
{
    return calloc(1, sizeof(fwd_table_t));
}

void table_destroy(fwd_table_t *table)
{
    if (table != NULL) {
        free(table->routes);
        free(table);
    }
}

size_t table_count(const fwd_table_t *table)
{
    return table == NULL ? 0 : table->count;
}

static bool next_route_capacity(size_t current, size_t *next)
{
    size_t capacity = current == 0 ? 4 : current * 2;
    if (capacity < current ||
        capacity > SIZE_MAX / sizeof(route_t)) {
        return false;
    }

    *next = capacity;
    return true;
}

bool table_add(fwd_table_t *table, const route_t *route)
{
    if (table == NULL || route == NULL ||
        route->dest.prefix > 32 || !cidr_is_network(route->dest) ||
        memchr(route->dev, '\0', sizeof(route->dev)) == NULL ||
        route->dev[0] == '\0') {
        return false;
    }

    // A destination/prefix pair may appear only once in the table.
    for (size_t index = 0; index < table->count; index++) {
        const cidr_t existing = table->routes[index].dest;
        if (existing.addr == route->dest.addr &&
            existing.prefix == route->dest.prefix) {
            return false;
        }
    }
    if (table->count == table->capacity) {
        // Grow geometrically, keeping the old allocation intact if realloc fails.
        size_t new_capacity;
        if (!next_route_capacity(table->capacity, &new_capacity)) { // GCOVR_EXCL_BR_LINE
            return false; // GCOVR_EXCL_LINE: only reachable at allocation-size exhaustion
        }
        route_t *new_routes = realloc(table->routes,
                                      new_capacity * sizeof(*table->routes));
        // Allocator failure is nondeterministic and originates from realloc().
        if (new_routes == NULL) // GCOVR_EXCL_START
        {
            return false;
        } // GCOVR_EXCL_STOP
        table->routes = new_routes;
        table->capacity = new_capacity;
    }

    table->routes[table->count++] = *route;
    return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line)
{
    if (bad_line != NULL) {
        *bad_line = 0;
    }
    if (table == NULL || in == NULL) {
        return false;
    }

    char *line = NULL;
    size_t capacity = 0;
    size_t line_number = 0;
    ssize_t length;
    while ((length = getline(&line, &capacity, in)) >= 0) {
        (void)length;
        line_number++;

        // Remove inline comments before deciding whether this line is blank or a route.
        char *comment = strchr(line, '#');
        if (comment != NULL) {
            *comment = '\0';
        }
        const char *cursor = line;
        skip_whitespace(&cursor);
        if (*cursor == '\0') {
            continue;
        }

        route_t route;
        if (!route_parse(cursor, &route) || !table_add(table, &route)) {
            if (bad_line != NULL) {
                *bad_line = line_number;
            }
            free(line);
            return false;
        }
    }

    bool read_error = ferror(in) != 0 || !feof(in); // GCOVR_EXCL_BR_LINE
    free(line);
    return !read_error;
}

const route_t *table_lookup(const fwd_table_t *table, uint32_t dest)
{
    if (table == NULL) {
        return NULL;
    }

    const route_t *best = NULL;
    for (size_t index = 0; index < table->count; index++) {
        const route_t *route = &table->routes[index];
        // Replacing best only for a longer prefix makes insertion order irrelevant.
        if (cidr_contains(route->dest, dest) &&
            (best == NULL || route->dest.prefix > best->dest.prefix)) {
            best = route;
        }
    }
    return best;
}

static bool parse_decimal(const char **text, uint32_t max, uint32_t *value)
{
    const char *cursor = *text;
    if (*cursor < '0' || *cursor > '9') {
        return false;
    }

    // Reject ambiguous octal-looking values such as "010".
    if (*cursor == '0' && cursor[1] >= '0' && cursor[1] <= '9') {
        return false;
    }

    uint32_t parsed = 0;
    do {
        uint32_t digit = (uint32_t)(*cursor - '0');
        // Check the next digit before multiplying, so oversized input cannot overflow.
        if (parsed > max / 10 ||
            (parsed == max / 10 && digit > max % 10)) {
            return false;
        }
        parsed = parsed * 10 + digit;
        cursor++;
    } while (*cursor >= '0' && *cursor <= '9');

    *text = cursor;
    *value = parsed;
    return true;
}

static bool parse_ipv4(const char **text, uint32_t *addr)
{
    const char *cursor = *text;
    uint32_t parsed = 0;

    // Parse exactly four bounded octets and pack them in network-style order.
    for (unsigned int octet_index = 0; octet_index < 4; octet_index++) {
        uint32_t octet;
        if (!parse_decimal(&cursor, UINT8_MAX, &octet)) {
            return false;
        }
        parsed = (parsed << 8) | octet;
        if (octet_index < 3) {
            if (*cursor != '.') {
                return false;
            }
            cursor++;
        }
    }

    *text = cursor;
    *addr = parsed;
    return true;
}

bool ip_parse(const char *text, uint32_t *addr)
{
    if (text == NULL || addr == NULL) {
        return false;
    }

    // Require the whole string to be consumed; only publish a result on success.
    const char *cursor = text;
    uint32_t parsed;
    if (!parse_ipv4(&cursor, &parsed) || *cursor != '\0') {
        return false;
    }

    *addr = parsed;
    return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
    // Extract the address's four octets from most to least significant.
    snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
        (unsigned int)((addr >> 24) & UINT32_C(0xff)),
        (unsigned int)((addr >> 16) & UINT32_C(0xff)),
        (unsigned int)((addr >> 8) & UINT32_C(0xff)),
        (unsigned int)(addr & UINT32_C(0xff)));
}

bool cidr_parse(const char *text, cidr_t *block) {
    if(text == NULL || block == NULL) {
        return false;
    }

    const char *cursor = text;
    cidr_t parsed;
    uint32_t prefix;
    // Parse an IPv4 address, then require a slash and a complete 0-32 prefix.
    if (!parse_ipv4(&cursor, &parsed.addr) || *cursor != '/') {
        return false;
    }
    cursor++;
    if (!parse_decimal(&cursor, 32, &prefix) || *cursor != '\0') {
        return false;
    }

    parsed.prefix = (uint8_t)prefix;
    *block = parsed;
    return true;
}

void cidr_format(cidr_t block, char out[CIDR_STR_LEN]) {
    char address[IP_STR_LEN];
    ip_format(block.addr, address);
    // Keep formatting consistent with cidr_mask(), which treats larger prefixes as /32.
    unsigned int prefix = block.prefix > 32 ? 32U : block.prefix;
    snprintf(out, CIDR_STR_LEN, "%s/%u", address, prefix);
}

uint32_t cidr_mask(uint8_t prefix)
{
    if (prefix == 0) {
        return 0;
    }
    if (prefix >= 32) {
        return UINT32_MAX;
    }
    // Handle /0 above so the shift count is always between 0 and 31.
    return UINT32_MAX << (32 - prefix);
}

uint32_t cidr_network(cidr_t block)
{
    return block.addr & cidr_mask(block.prefix);
}

uint32_t cidr_broadcast(cidr_t block)
{
    return block.addr | ~cidr_mask(block.prefix);
}

uint32_t cidr_first_host(cidr_t block)
{
    uint32_t network = cidr_network(block);
    // /31 and /32 have no separate network address to skip.
    return block.prefix >= 31 ? network : network + 1;
}

uint32_t cidr_last_host(cidr_t block)
{
    uint32_t broadcast = cidr_broadcast(block);
    // /31 and /32 have no separate broadcast address to skip.
    return block.prefix >= 31 ? broadcast : broadcast - 1;
}

uint64_t cidr_host_count(cidr_t block)
{
    if (block.prefix >= 32) {
        return 1;
    }
    if (block.prefix == 31) {
        return 2;
    }
    // Use 64 bits so the /0 address count (2^32) is representable before subtracting.
    return (UINT64_C(1) << (32 - block.prefix)) - 2;
}

bool cidr_contains(cidr_t block, uint32_t addr)
{
    uint32_t mask = cidr_mask(block.prefix);
    return (addr & mask) == (block.addr & mask);
}

bool cidr_is_network(cidr_t block)
{
    return block.addr == cidr_network(block);
}

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
    if (out == NULL || block.prefix > 32 || !cidr_is_network(block) ||
        n == 0 || (n & (n - 1)) != 0 || i >= n) {
        return false;
    }

    // A power-of-two subnet count borrows log2(n) bits from the host portion.
    uint8_t subnet_bits = 0;
    for (uint32_t count = n; count > 1; count >>= 1) {
        subnet_bits++;
    }
    uint16_t new_prefix = (uint16_t)block.prefix + subnet_bits;
    if (new_prefix > 32) {
        return false;
    }

    // Keep /0's possible 2^32-address size representable during offset arithmetic.
    uint64_t subnet_size = UINT64_C(1) << (32 - new_prefix);
    uint64_t subnet_addr = (uint64_t)block.addr + subnet_size * i;
    cidr_t subnet = { (uint32_t)subnet_addr, (uint8_t)new_prefix };
    *out = subnet;
    return true;
}

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H

#define _POSIX_C_SOURCE 200809L
#include "lab.h"

#include <inttypes.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static void print_usage(FILE *stream)
{
    fputs("Usage: myapp info <address>/<prefix>\n"
          "       myapp split <network>/<prefix> <count>\n"
          "       myapp lookup <table-file> <address>...\n",
          stream);
}

static bool parse_count(const char *text, uint32_t *count)
{
    if (text == NULL || *text == '\0') {
        return false;
    }

    uint32_t parsed = 0;
    for (const char *cursor = text; *cursor != '\0'; cursor++) {
        // Accept decimal digits only and detect overflow before multiplication.
        if (*cursor < '0' || *cursor > '9') {
            return false;
        }
        uint32_t digit = (uint32_t)(*cursor - '0');
        if (parsed > UINT32_MAX / 10 ||
            (parsed == UINT32_MAX / 10 && digit > UINT32_MAX % 10)) {
            return false;
        }
        parsed = parsed * 10 + digit;
    }

    *count = parsed;
    return true;
}

static void format_address(uint32_t address, char formatted[IP_STR_LEN])
{
    ip_format(address, formatted);
}

static int command_info(const char *text)
{
    cidr_t block;
    if (!cidr_parse(text, &block)) {
        fprintf(stderr, "myapp: '%s' is not a valid address/prefix\n", text);
        return 1;
    }

    char address[IP_STR_LEN];
    char netmask[IP_STR_LEN];
    char network[IP_STR_LEN];
    char broadcast[IP_STR_LEN];
    char first_host[IP_STR_LEN];
    char last_host[IP_STR_LEN];
    format_address(block.addr, address);
    // Derive all displayed subnet values from the parsed address and prefix.
    format_address(cidr_mask(block.prefix), netmask);
    format_address(cidr_network(block), network);
    format_address(cidr_broadcast(block), broadcast);
    format_address(cidr_first_host(block), first_host);
    format_address(cidr_last_host(block), last_host);

    printf("address    %s\n"
           "prefix     /%u\n"
           "netmask    %s\n"
           "network    %s\n"
           "broadcast  %s\n"
           "first host %s\n"
           "last host  %s\n"
           "hosts      %" PRIu64 "\n",
           address, (unsigned int)block.prefix, netmask, network, broadcast,
           first_host, last_host, cidr_host_count(block));
    return 0;
}

static int command_split(const char *text, const char *count_text)
{
    cidr_t block;
    uint32_t count;
    if (!cidr_parse(text, &block)) {
        fprintf(stderr, "myapp: '%s' is not a valid network/prefix\n", text);
        return 1;
    }
    if (!parse_count(count_text, &count) || count == 0) {
        fprintf(stderr, "myapp: '%s' is not a valid subnet count\n", count_text);
        return 1;
    }
    if (!cidr_is_network(block)) {
        fprintf(stderr, "myapp: '%s' has host bits set; expected a network\n", text);
        return 1;
    }

    // Validate the split once before printing, so invalid requests produce no partial output.
    cidr_t subnet;
    if (!cidr_subnet(block, count, 0, &subnet)) {
        fprintf(stderr, "myapp: cannot split '%s' into %" PRIu32 " equal subnets\n",
                text, count);
        return 1;
    }

    for (uint32_t index = 0; index < count; index++) {
        if (!cidr_subnet(block, count, index, &subnet)) {
            fprintf(stderr, "myapp: failed to calculate subnet %" PRIu32 "\n", index);
            return 1;
        }
        char formatted[CIDR_STR_LEN];
        cidr_format(subnet, formatted);
        puts(formatted);
    }
    return 0;
}

static int command_lookup(const char *path, int address_count, char *addresses[])
{
    FILE *input = fopen(path, "r");
    if (input == NULL) {
        fprintf(stderr, "myapp: cannot open %s: %s\n", path, strerror(errno));
        return 2;
    }

    fwd_table_t *table = table_create();
    if (table == NULL) {
        fprintf(stderr, "myapp: cannot create forwarding table\n");
        fclose(input);
        return 2;
    }

    size_t bad_line;
    if (!table_load(table, input, &bad_line)) {
        if (bad_line != 0) {
            fprintf(stderr, "myapp: %s:%zu: not a valid route\n", path, bad_line);
        } else {
            fprintf(stderr, "myapp: could not read %s\n", path);
        }
        table_destroy(table);
        fclose(input);
        return 2;
    }
    fclose(input);

    // Continue through every argument; malformed addresses take precedence over unreachable ones.
    bool malformed = false;
    bool unreachable = false;
    for (int index = 0; index < address_count; index++) {
        uint32_t address;
        if (!ip_parse(addresses[index], &address)) {
            fprintf(stderr, "myapp: '%s' is not a valid IPv4 address\n",
                    addresses[index]);
            malformed = true;
            continue;
        }

        char formatted[IP_STR_LEN];
        format_address(address, formatted);
        const route_t *route = table_lookup(table, address);
        if (route == NULL) {
            printf("%s unreachable\n", formatted);
            unreachable = true;
        } else if (route->gateway != 0) {
            char gateway[IP_STR_LEN];
            format_address(route->gateway, gateway);
            printf("%s via %s dev %s\n", formatted, gateway, route->dev);
        } else {
            printf("%s dev %s\n", formatted, route->dev);
        }
    }

    table_destroy(table);
    if (malformed) {
        return 1;
    }
    return unreachable ? 3 : 0;
}

int main(int argc, char *argv[])
{
    // The no-argument invocation is a successful usage path (used by make leak).
    if (argc == 1) {
        print_usage(stdout);
        return 0;
    }

    if (strcmp(argv[1], "info") == 0) {
        if (argc != 3) {
            print_usage(stderr);
            return 1;
        }
        return command_info(argv[2]);
    }

    if (strcmp(argv[1], "split") == 0) {
        if (argc != 4) {
            print_usage(stderr);
            return 1;
        }
        return command_split(argv[2], argv[3]);
    }

    if (strcmp(argv[1], "lookup") == 0) {
        if (argc < 4) {
            print_usage(stderr);
            return 1;
        }
        return command_lookup(argv[2], argc - 3, &argv[3]);
    }

    fprintf(stderr, "myapp: unknown command '%s'\n", argv[1]);
    print_usage(stderr);
    return 1;
}

void test_forwarding_table(void)
{

        "132.178.227.0/25 direct eno1\n"
        "192.168.0.0/16 10.1.0.1 eth1\n", input);
  rewind(input);

  bad_line = 99;
  TEST_ASSERT_TRUE(table_load(table, input, &bad_line));
  TEST_ASSERT_EQUAL_size_t(0, bad_line);
  TEST_ASSERT_EQUAL_size_t(7, table_count(table));
  fclose(input);

  TEST_ASSERT_TRUE(match != NULL);
  TEST_ASSERT_EQUAL_UINT8(25, match->dest.prefix);
  TEST_ASSERT_EQUAL_STRING("eth2", match->dev);

  match = table_lookup(table, UINT32_C(0x0a010250));
  TEST_ASSERT_TRUE(match != NULL);
  TEST_ASSERT_EQUAL_UINT8(24, match->dest.prefix);
  TEST_ASSERT_EQUAL_STRING("eth2", match->dev);

  match = table_lookup(table, UINT32_C(0x0a010350));
  TEST_ASSERT_TRUE(match != NULL);
  TEST_ASSERT_EQUAL_UINT8(16, match->dest.prefix);
  TEST_ASSERT_EQUAL_STRING("eth1", match->dev);

  match = table_lookup(table, UINT32_C(0xc0a80101));
  TEST_ASSERT_TRUE(match != NULL);
  TEST_ASSERT_EQUAL_UINT8(16, match->dest.prefix);
  TEST_ASSERT_EQUAL_STRING("eth1", match->dev);

  match = table_lookup(table, UINT32_C(0x08080808));
  TEST_ASSERT_TRUE(match != NULL);
  TEST_ASSERT_EQUAL_UINT8(0, match->dest.prefix);
  TEST_ASSERT_EQUAL_STRING("eno1", match->dev);

  invalid = (route_t){ { UINT32_C(0x0a010000), 33 }, 0, "eth9" };

  TEST_ASSERT_FALSE(table_add(table, &invalid));
  memset(invalid.dev, 'x', sizeof(invalid.dev));
  invalid.dest = (cidr_t){ 0, 0 };
  TEST_ASSERT_FALSE(table_add(table, &invalid));
  TEST_ASSERT_FALSE(table_add(table, &duplicate));
  TEST_ASSERT_FALSE(table_add(NULL, &duplicate));
  TEST_ASSERT_FALSE(table_add(table, NULL));
  TEST_ASSERT_EQUAL_size_t(7, table_count(table));

  input = tmpfile();
  TEST_ASSERT_TRUE(input != NULL);
  fputs("# first valid line\n"
        "192.0.2.0/24 direct eth0\n"
        "\n"
        "not-a-cidr direct eth1\n"
        "192.0.3.0/24 direct eth0\n", input);
  rewind(input);
  TEST_ASSERT_FALSE(table_load(table, input, &bad_line));
  TEST_ASSERT_EQUAL_size_t(4, bad_line);
  TEST_ASSERT_EQUAL_size_t(8, table_count(table));
  fclose(input);

  input = tmpfile();
  TEST_ASSERT_TRUE(input != NULL);
  fputs("192.0.2.0/24 direct eth0\n", input);

  TEST_ASSERT_FALSE(table_load(table, input, NULL));
  fclose(input);

  fputs("198.51.100.0/24 direct eth0\n", input);
  rewind(input);
  TEST_ASSERT_TRUE(table_load(table, input, NULL));
  TEST_ASSERT_EQUAL_size_t(9, table_count(table));
  fclose(input);

  table_destroy(table);
  table_destroy(NULL);
}
